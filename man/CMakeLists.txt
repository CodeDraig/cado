file(GLOB CADO_MAN_PAGES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.[1-8]")
foreach(CADO_MAN_PATH IN LISTS CADO_MAN_PAGES)
	get_filename_component(CADO_MAN_PAGE "${CADO_MAN_PATH}" NAME)
	string(REGEX REPLACE ".*\\." "" CADO_MAN_SECTION "${CADO_MAN_PAGE}")
	install(FILES "${CADO_MAN_PATH}"
		DESTINATION "${CMAKE_INSTALL_MANDIR}/man${CADO_MAN_SECTION}")
endforeach()
